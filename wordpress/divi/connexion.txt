      cours_texte: 'Si vous suivez un cours Genitem, créez votre compte vous-même pendant la séance.',

En dehors, demandez-en un par le <a class="lien-contact" href="https://genitem.recherche-formation.com/contact/"><svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.4" stroke-linecap="round" stroke-linejoin="round" aria-hidden="true" style="vertical-align:-2px; margin-right:5px;"><rect x="2.5" y="4.5" width="19" height="15" rx="2"></rect><polyline points="3 6 12 13 21 6"></polyline></svg>formulaire de contact</a>.'
    },
    en: {
      titre: 'Log in',
      chapeau: 'Logging in to Genitem gives you access to two things:',
      point_forum: 'the <strong>forum</strong>, to ask a question or share your experience;',
      point_modeles: 'your <strong>saved models</strong>: the models you add or remove stay recorded on your account, and you find them again on every visit, with nothing to download or reinstall.',
      sans_compte: 'Without logging in, the pages remain fully readable and usable: only the saving of your models is unavailable.',
      cours_titre: 'Signing up during a lesson',
      cours_texte: 'If you are taking a Genitem course, create your own account during the session.',
      cours_lien: 'Registration form',
      note: 'Creating an account is reserved for course participants. Otherwise, request one through the <a class="lien-contact" href="https://genitem.recherche-formation.com/contact/"><svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.4" stroke-linecap="round" stroke-linejoin="round" aria-hidden="true" style="vertical-align:-2px; margin-right:5px;"><rect x="2.5" y="4.5" width="19" height="15" rx="2"></rect><polyline points="3 6 12 13 21 6"></polyline></svg>contact form</a>.'
    }
  };

  var bloc = document.getElementById('introConnexion');

  window.introLangue = function (langue) {
    var dict = TEXTES[langue] || TEXTES.fr;

    bloc.querySelectorAll('[data-i18n]').forEach(function (el) {
      var cle = el.getAttribute('data-i18n');
      if (dict[cle] !== undefined) el.innerHTML = dict[cle];
    });
    bloc.querySelectorAll('[data-i18n-html]').forEach(function (el) {
      var cle = el.getAttribute('data-i18n-html');
      if (dict[cle] !== undefined) el.innerHTML = dict[cle];
    });

    document.getElementById('introBtnFr').classList.toggle('langue-active', langue === 'fr');
    document.getElementById('introBtnEn').classList.toggle('langue-active', langue === 'en');

    // Memorise le choix, comme sur les autres pages du site
    try { localStorage.setItem('langueConfigurateur', langue); } catch (e) {}
  };

  // Langue de depart : preference enregistree, sinon celle du navigateur
  var depart = 'fr';
  try {
    var memorisee = localStorage.getItem('langueConfigurateur');
    if (memorisee === 'fr' || memorisee === 'en') depart = memorisee;
    else depart = (navigator.language || 'fr').toLowerCase().indexOf('fr') === 0 ? 'fr' : 'en';
  } catch (e) {}

  window.introLangue(depart);
})();
</script>
