<script>
(function () {
  var menu = document.getElementById('menuGenitem');
  var bouton = menu.querySelector('.menu-burger');

  // Divi enferme le menu dans plusieurs conteneurs qui peuvent rogner le
  // sous-menu deroulant ou le faire passer sous la section suivante. On
  // remonte la chaine des parents pour lever ces deux obstacles. Le faire
  // par script plutot que par feuille de style evite de dependre du
  // selecteur :has() et des noms de classes de Divi.
  (function libererConteneurs(){
    var el = menu.parentNode;
    while (el && el.nodeType === 1 && el !== document.body) {
      // Sans condition : ces conteneurs entourent le menu et n'ont aucune
      // raison de defiler ni de masquer quoi que ce soit.
      el.style.overflow = 'visible';
      if (window.getComputedStyle(el).position === 'static') el.style.position = 'relative';
      el.style.zIndex = '9000';
      el = el.parentNode;
    }
  })();

  bouton.addEventListener('click', function () {
    var ouvert = menu.classList.toggle('ouvert');
    bouton.setAttribute('aria-expanded', ouvert ? 'true' : 'false');
  });

  // Sur ecran tactile, un premier clic sur « Extra » ouvre le sous-menu ;
  // un second suit le lien vers la page de presentation.
  menu.querySelectorAll('.menu-deroulant').forEach(function (bloc) {
    var parent = bloc.querySelector('.menu-parent');
    parent.addEventListener('click', function (e) {
      var tactile = window.matchMedia('(hover: none)').matches
        || window.matchMedia('(max-width: 1024px)').matches;
      if (tactile && !bloc.classList.contains('ouvert')) {
        e.preventDefault();
        bloc.classList.add('ouvert');
        parent.setAttribute('aria-expanded', 'true');
      }
    });
  });

  // Referme le menu quand on choisit un lien, ou en cliquant a cote
  menu.querySelectorAll('nav a').forEach(function (lien) {
    lien.addEventListener('click', function () {
      menu.classList.remove('ouvert');
      bouton.setAttribute('aria-expanded', 'false');
      menu.querySelectorAll('.menu-deroulant').forEach(function (b) { b.classList.remove('ouvert'); });
    });
  });
  document.addEventListener('click', function (e) {
    if (!menu.contains(e.target)) {
      menu.classList.remove('ouvert');
      bouton.setAttribute('aria-expanded', 'false');
    }
  });

  // Etat de connexion : depose par le plugin sur la page WordPress. Depuis un
  // cadre, on le lit chez le parent, qui est du meme domaine.
  function contexte() {
    try { if (window.parent && window.parent.GenitemAuth) return window.parent.GenitemAuth; } catch (e) {}
    return window.GenitemAuth || null;
  }

  var cadenas = document.getElementById('menuConnexion');
  var ctx = contexte();

  // L'entree « Extra » n'apparait que pour une session ouverte.
  if (ctx && ctx.connecte) menu.classList.add('connecte');

  if (cadenas && ctx && ctx.connecte) {
    cadenas.classList.add('connecte');
    cadenas.querySelector('.anse-fermee').style.display = 'none';
    cadenas.querySelector('.anse-ouverte').style.display = '';
    cadenas.querySelector('.cadenas-nom').textContent = ctx.nom || '';
    cadenas.setAttribute('title', 'Connecté : ' + (ctx.nom || ''));
    cadenas.setAttribute('href', ctx.urlLogout || cadenas.getAttribute('href'));
  }

  // Met en surbrillance le lien de la page actuelle
  menu.querySelectorAll('nav a[href]').forEach(function (lien) {
    if (location.href.indexOf(lien.getAttribute('href')) === 0) {
      lien.classList.add('active');
    }
  });
})();
</script>
